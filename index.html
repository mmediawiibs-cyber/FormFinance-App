Pengaturan Web App URL & Kode `Code.gs` (23 Kolom)

Terhubung dengan Google Sheet: Form Finance STIT (1qdxGHKNCSkuhEccI_W63ShWv-bFuKVsTK09nrYdqdnw).

Rumus VLOOKUP pada Sheet "Form PP": Gunakan rentang Data PP!A2:W100 
=IF((VLOOKUP($A$1, 'Data PP'!$A$2:$W$100, 11, FALSE))<>"", VLOOKUP($A$1, 'Data PP'!$A$2:$W$100, 11, FALSE), "")

Catatan: Ganti angka kolom ke-11 pada rumus di atas sesuai dengan kolom yang ingin Anda tampilkan dari tabel Data PP (Contoh: 3 untuk Nomor PP, 4 untuk Pembuat, 16 untuk Nominal, dll). 

var SPREADSHEET_ID = "1qdxGHKNCSkuhEccI_W63ShWv-bFuKVsTK09nrYdqdnw";
var ss = SpreadsheetApp.openById(SPREADSHEET_ID);

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheetData = ss.getSheetByName("Data PP");
    
    // Simpan lampiran ke Google Drive jika ada file yang diunggah
    var lampiranLink = data.lampiran || "Terlampir";
    if (data.fileLampiran && data.fileLampiran.data) {
      var blob = Utilities.newBlob(Utilities.base64Decode(data.fileLampiran.data), data.fileLampiran.mimeType, data.fileLampiran.name);
      var file = DriveApp.createFile(blob);
      lampiranLink = file.getUrl();
    }

    // Jika aksi update/edit baris
    if (data.action === "updateData") {
      var rows = sheetData.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        if (Number(rows[i][0]) === Number(data.nomor)) {
          var r = i + 1;
          sheetData.getRange(r, 8).setValue(data.uraian1);
          sheetData.getRange(r, 16).setValue(data.nominal);
          sheetData.getRange(r, 18).setValue(data.tanggalEntry);
          sheetData.getRange(r, 19).setValue(data.status);
          sheetData.getRange(r, 20).setValue(data.tanggalTf);
          sheetData.getRange(r, 21).setValue(data.jumlahTf);
          sheetData.getRange(r, 22).setValue(data.tanggalRekap);
          sheetData.getRange(r, 23).setValue(data.keterangan);
          return ContentService.createTextOutput(JSON.stringify({status:"success", message:"Data diperbarui"})).setMimeType(ContentService.MimeType.JSON);
        }
      }
    }

    // Susunan 23 Kolom Sheet "Data PP"
    // 11: KEPADA, 12: SEBAGAI, 13: A/N PENERIMA, 14: BANK, 15: NO. REKENING
    var rowData = [
      data.nomor,                // 1. NO
      data.divisi,               // 2. PP (Kode Divisi)
      data.nomorPp,              // 3. NOMOR PP
      data.namaPembuat,          // 4. NAMA PEMBUAT
      data.tanggalPp,            // 5. TANGGAL PP
      data.tanggalPemenuhan,     // 6. TANGGAL PEMENUHAN
      data.tujuan,               // 7. TUJUAN
      data.uraian1,              // 8. URAIAN I (Tujuan)
      data.uraian2,              // 9. URAIAN II (Objek)
      data.uraian3,              // 10. URAIAN III (Keterangan)
      data.kepada,               // 11. KEPADA
      data.sebagai,              // 12. SEBAGAI
      data.anPenerima,           // 13. A/N PENERIMA
      data.bank,                 // 14. BANK
      data.noRekening,           // 15. NO. REKENING
      data.nominal,              // 16. NOMINAL
      lampiranLink,              // 17. LAMPIRAN
      data.tanggalEntry,         // 18. TANGGAL ENTRY
      data.status || "NOT PAID", // 19. STATUS
      data.tanggalTf || "",      // 20. TANGGAL TF
      data.jumlahTf || "",       // 21. JUMLAH TF
      data.tanggalRekap || "",   // 22. TANGGAL REKAP TF
      data.keterangan || ""      // 23. KETERANGAN
    ];

    sheetData.appendRow(rowData);
    return ContentService.createTextOutput(JSON.stringify({status:"success", nomorUrut: data.nomor, nomorPp: data.nomorPp})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status:"error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var action = e.parameter.action;
  var sheetData = ss.getSheetByName("Data PP");
  var sheetForm = ss.getSheetByName("Form PP");

  if (action === "getAllData") {
    var rows = sheetData.getDataRange().getValues();
    var list = [];
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] !== "") {
        list.push({
          nomor: Number(rows[i][0]), divisi: rows[i][1], nomorPp: rows[i][2],
          namaPembuat: rows[i][3], tanggalPp: rows[i][4], tanggalPemenuhan: rows[i][5],
          tujuan: rows[i][6], uraian1: rows[i][7], uraian2: rows[i][8], uraian3: rows[i][9],
          kepada: rows[i][10], sebagai: rows[i][11], anPenerima: rows[i][12],
          bank: rows[i][13], noRekening: rows[i][14], nominal: rows[i][15],
          lampiran: rows[i][16], tanggalEntry: rows[i][17], status: rows[i][18] || "NOT PAID",
          tanggalTf: rows[i][19], jumlahTf: rows[i][20], tanggalRekap: rows[i][21], keterangan: rows[i][22]
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify(list)).setMimeType(ContentService.MimeType.JSON);
  }

  if (action === "cetakPDF" && sheetForm) {
    sheetForm.getRange("A1").setValue(e.parameter.nomor);
    SpreadsheetApp.flush();
    var url = "https://docs.google.com/spreadsheets/d/" + ss.getId() + "/export?exportFormat=pdf&format=pdf&size=7&portrait=true&fitw=true&gid=" + sheetForm.getSheetId();
    return HtmlService.createHtmlOutput("<script>window.location.href='" + url + "';</script>");
  }
}
